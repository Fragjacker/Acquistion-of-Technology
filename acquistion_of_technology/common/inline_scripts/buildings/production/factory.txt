inline_script = {
    script = shared/normal_buildings_base
    ALLOW = "$ALLOW$"
    CAN_BUILD = $CAN_BUILD$
    PREREQUISITE = $PREREQUISITE$
    UPGRADES_INTO = $UPGRADES_INTO$
    AI_FACTORS = "$AI_FACTORS$"
    TIER = $TIER$
}
@num_jobs = @[ 200 * $TIER$ ]
category = manufacturing
building_sets = {
    industrial
    factory
    aot_building_tier_$TIER$
}
potential = {
    owner? = {
        OR = {
            country_uses_consumer_goods = yes
            has_origin = origin_fear_of_the_dark
        }
        has_technology = $PREREQUISITE$
    }
    OR = {
        has_carrier_flag = ignore_ai_building_limitations
        owner = {
            is_ai = no
        }

        # Obsessional machines can build these freely. Have fun paperclipping.
        owner = {
            has_valid_civic = civic_machine_obsessional_directive
        }

        # AI is allowed to build one of these if they have a FACTORY specialization - intentionally more restrictive than foundries.
        AND = {
            has_any_factory_zone = yes
            NOR = {
                has_building = building_factory_1
                has_building = building_factory_2
                has_building = building_factory_3
            }
        }
    }
}
country_modifier = {
    country_resource_max_consumer_goods_add = @[5000 * two_exp$EXPONENT$]
}
triggered_planet_modifier = {
    potential = {
        owner? = {
            is_individualistic_empire = yes
        }
    }
    modifier = {
        job_$JOB$_add = @num_jobs
    }
}
triggered_planet_modifier = {
    potential = {
        owner? = {
            is_gestalt = yes
            country_uses_consumer_goods = yes
        }
    }
    modifier = {
        job_$JOB_HIVE$_add = @num_jobs
    }
}
resources = {
    category = planet_buildings
    cost = {
        $COST$
    }
    produces = {
        $PRODUCES$
    }
    produces = {
        trigger = {
            has_modifier = acot_modifier_ascended_city
        }
        consumer_goods = 40
    }
    upkeep = {
        trigger = {
            has_modifier = acot_modifier_ascended_city
        }
        sr_dark_matter = 2
        acot_sr_dark_energy = 2
        minerals = 20
    }
}
triggered_desc = {
    trigger = {
        owner? = {
            is_individualistic_empire = yes
        }
    }
    text = job_artisan_effect_desc
}
triggered_desc = {
    trigger = {
        owner? = {
            is_gestalt = yes
            country_uses_consumer_goods = yes
        }
    }
    text = job_artisan_drone_effect_desc
}
