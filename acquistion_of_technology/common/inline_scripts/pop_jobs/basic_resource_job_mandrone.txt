###########
# makes jobs of every level in both individual and gestalt forms
#
# REQUIREMENTS:
# the job fulfills man and drone suffix
# the acot job must be delta equivalent
###########


acot_$resource$man = {
    inline_script = {
        script = pop_jobs/part/specialist
        JOB = acot_$resource$man
        RESOURCE = $resource_plural$
    }

    swappable_data = {
        default = {
            condition_string = SPECIALIST_JOB_TRIGGER
            building_icon = building_acot_dm_$building_string$
            icon = $icon_dm$
        }
    }

    tags = { $resource_plural$ }

    resources = {
        category = $planet_category$
        produces = {
            $resource_plural$ = @aot_$resource$_DM_prod_base
        }

        upkeep = {
            sr_dark_matter = @aot_$resource$_DM_upkeep_matter
        }
    }

    inline_script = {
        script = generic_parts/aot_toggled_code
        TOGGLE = $IS_TECHNICIAN$
        CODE = "
        triggered_planet_modifier = {
            potential = {
                technician_is_fuel_chemist_trigger = yes
            }
            ship_speed_mult = @[0.05 * two_exp0]
        }

        triggered_planet_modifier = {
            potential = {
                technician_is_shield_tuner_trigger = yes
            }
            ship_shield_mult = @[0.02 * two_exp0] #move ship bonuses on technician jobs, add a custom tooltip here
            ship_shield_regen_mult = @[0.02 * two_exp0]
        }
        "
    }

    inline_script = "jobs/automodding_priority_$resource_plural$"
}
